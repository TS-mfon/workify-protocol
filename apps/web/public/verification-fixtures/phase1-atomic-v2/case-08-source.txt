IMPLEMENTATION (JavaScript)

export function slugifyTitle(input) {
  const value = String(input).trim().toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
  if (!value) throw new TypeError("empty slug");
  return value;
}
