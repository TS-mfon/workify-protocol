export function parsePort(input) {
  if (!/^[0-9]+$/.test(String(input))) throw new TypeError("integer required");
  const port = Number(input);
  if (!Number.isInteger(port) || port < 1 || port > 65535) throw new RangeError("invalid port");
  return port;
}
