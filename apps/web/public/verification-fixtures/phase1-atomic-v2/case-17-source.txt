```js
export function dedupeTags(tags) {
  const seen = new Set();
  const result = [];
  for (const item of tags) {
    const tag = String(item).trim().toLowerCase();
    if (tag && !seen.has(tag)) { seen.add(tag); result.push(tag); }
  }
  return result;
}
```
