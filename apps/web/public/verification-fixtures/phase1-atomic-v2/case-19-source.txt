export function safeFilename(input) {
  const value = String(input).replace(/[\\/]+/g, "").replace(/\s+/g, " ").trim().slice(0, 80);
  if (!value || /^\.+$/.test(value)) throw new TypeError("unsafe filename");
  return value;
}
