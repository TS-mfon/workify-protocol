```js
export function normalizeEmail(input) {
  const value = String(input).trim();
  const match = /^([^@\s]+)@([^@\s]+\.[^@\s]+)$/.exec(value);
  if (!match) throw new TypeError("invalid email");
  return match[1] + "@" + match[2].toLowerCase();
}
```
