IMPLEMENTATION (JavaScript)
```js
export function normalizeAddress(input) {
  const value = String(input).replace(/^0x/i, "");
  if (!/^[0-9a-fA-F]{40}$/.test(value)) throw new TypeError("invalid address");
  return "0x" + value.toLowerCase();
}
```
